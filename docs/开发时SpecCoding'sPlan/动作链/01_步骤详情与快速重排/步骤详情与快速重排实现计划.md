# 步骤详情与快速重排 — 实现计划

> **执行方式：** Inline Execution（本会话 executing-plans，批量 + 检查点）。不派子代理（遵守用户的串行 / 禁 Agent Teams / 禁后台规则）。每个有逻辑的任务走 TDD（先写失败测试 → 实现 → 通过）。

**Goal:** 修复 Qt 步骤详情漏渲染 describe 的 bug，详情面板补齐「摘要 + 关键参数表 + 全部字段 + 序号输入移动 + 复制」，列表新增「拖拽 / 多选批量 / 置顶置底」快速重排，Qt/tk 完全统一。

**Architecture:** 业务层（ChainModel / Controller / 关键参数配置 / 字段格式化）共用，纯渲染 Qt/tk 各自实现但规格一致。新增 insert 语义的 `reorder_steps` 作为拖拽 / 置顶置底 / 目标序号 / 批量移动的统一底层（既有 `move_step` 是 swap 语义，仅保留给 ↑↓ 相邻移动）。

**Tech Stack:** Python 3.11+，PySide6（Qt），tkinter（tk），pytest。

## Global Constraints

- 不可变优先；函数 <50 行；文件聚焦。
- 命名禁空格（路径 / 文件名）。
- Qt/tk 共用层单一事实来源；UI 严格按统一规格表。
- `reorder_steps` 只换 ACTION 节点内容，**不动 DAG 边**。
- 执行中（`_require_idle`）所有写操作抛 `RuntimeError`，UI 给提示不崩溃。
- 测试覆盖率 ≥80%；新增 i18n key 同步 zh/en。

```python
def reorder_steps(self, new_order: list[int]) -> None:
    """按 new_order（原索引的新排列）insert 语义重排所有 ACTION 节点内容。

    new_order[i] = 新序列位置 i 应承载的原步骤索引。
    仅交换节点承载的 action/comment/enabled，不动 DAG 边。
    """
    action_nodes = self.graph.action_nodes()
    n = len(action_nodes)
    if len(new_order) != n or sorted(new_order) != list(range(n)):
        return  # 非合法排列，忽略
    old = [(nd.action, nd.comment, nd.enabled) for nd in action_nodes]
    for node, old_idx in zip(action_nodes, new_order):
        action, comment, enabled = old[old_idx]
        node.action = action
        node.comment = comment
        node.enabled = enabled
    self._mark_dirty()
    self._bus.emit(EventName.CHAIN_STEPS_CHANGED)

def duplicate_step(self, index: int) -> int:
    """深拷贝步骤，副本插入到 index 之后，返回副本新索引；越界返回 -1。"""
    import copy
    steps = self.get_steps()
    if not (0 <= index < len(steps)):
        return -1
    self.add_step(copy.deepcopy(steps[index]))  # 追加到末尾
    n = len(self.get_steps())
    new_order = list(range(n))
    last = new_order.pop()           # 副本当前在末尾
    new_order.insert(index + 1, last)  # 移到 index+1
    self.reorder_steps(new_order)
    return index + 1
```

def reorder_steps(self, new_order: list[int]) -> None:
    self._require_idle()
    self.model.reorder_steps(new_order)

def duplicate_step(self, index: int) -> int:
    self._require_idle()
    return self.model.duplicate_step(index)

def move_to_index(self, index: int, target: int) -> None:
    """把 index 处步骤 insert 移动到 target，其余顺延。"""
    self._require_idle()
    steps = self.model.get_steps()
    n = len(steps)
    if not (0 <= index < n and 0 <= target < n) or index == target:
        return
    order = list(range(n))
    moved = order.pop(index)
    order.insert(target, moved)
    self.model.reorder_steps(order)

```python
"""步骤参数视图共用工具：重排 order 构造 + 字段值格式化。Qt/tk 共用。"""
from __future__ import annotations
import dataclasses
import os
from enum import Enum

from src.utils.i18n import t


def build_move_order(n: int, src: int, dst: int) -> list[int]:
    """把 src 取出 insert 到 dst，其余顺延；返回原索引的新排列。"""
    if not (0 <= src < n and 0 <= dst < n) or src == dst:
        return list(range(n))
    order = list(range(n))
    moved = order.pop(src)
    order.insert(dst, moved)
    return order


def build_batch_move_order(n: int, selected: list[int], delta: int) -> list[int]:
    """选中索引集合整体 ±1 移动（跳过已在边界的）。"""
    sel = set(selected)
    order = list(range(n))
    if delta < 0:
        indices = sorted(sel)
        for i in indices:
            j = i - 1
            if j < 0 or j in sel:
                continue
            order[i], order[j] = order[j], order[i]
    elif delta > 0:
        for i in sorted(sel, reverse=True):
            j = i + 1
            if j >= n or j in sel:
                continue
            order[i], order[j] = order[j], order[i]
    return order


def format_field_value(step, field_name: str) -> str:
    """格式化单个字段值为人类可读字符串。"""
    value = getattr(step, field_name, None)
    if value is None:
        return "--"
    if isinstance(value, Enum):
        return getattr(value, "name", str(value))
    if isinstance(value, bool):
        return "✓" if value else "✗"
    if isinstance(value, str) and (value == "" or "/" in value or "\\" in value):
        return os.path.basename(value) if value else t("common.not_set")
    if isinstance(value, (list, tuple)):
        return t("chain.detail.n_items", count=len(value)) if value else "--"
    return str(value)


def iter_all_fields(step):
    """遍历 dataclass 全部字段（排除 ClassVar），yield (字段名, 格式化值)。"""
    for f in dataclasses.fields(step):
        if f._field_type is dataclasses._FIELD:  # 跳过 ClassVar
            yield (f.name, format_field_value(step, f.name))
```

> 实现时校准 `dataclasses.fields` 对带 `ClassVar` 的 dataclass 的行为（必要时用 `getattr` 过滤）。`chain.detail.n_items` i18n key 在 Task 4 加。
